import { Router } from 'express';
import { z } from 'zod';
import { Task } from '../models/Task';
import { User } from '../models/User';
import { AppError } from '../middleware/error';
import { requireAuth } from '../middleware/auth';
import { objectId } from '../utils/ids';
import { localDayBounds } from '../utils/time';
import { completeTask, toTaskDtos } from '../services/tasks';

export const tasksRouter = Router();
tasksRouter.use(requireAuth);

const listQuery = z.object({
  scope: z.enum(['overdue', 'today', 'upcoming', 'all']).default('all'),
  plantId: objectId.optional(),
  limit: z.coerce.number().int().min(1).max(100).default(50),
});

tasksRouter.get('/', async (req, res) => {
  const userId = req.userId!;
  const { scope, plantId, limit } = listQuery.parse(req.query);

  const user = await User.findById(userId).select('timezone');
  const { start, end } = localDayBounds(user?.timezone ?? 'UTC');

  const due =
    scope === 'overdue'
      ? { $lt: start }
      : scope === 'today'
        ? { $gte: start, $lt: end }
        : scope === 'upcoming'
          ? { $gte: end }
          : undefined;

  const tasks = await Task.find({
    userId,
    status: 'pending',
    ...(due && { dueAt: due }),
    ...(plantId && { plantId }),
  })
    .sort({ dueAt: 1 })
    .limit(limit);

  res.json({ tasks: await toTaskDtos(userId, tasks) });
});

tasksRouter.post('/:id/complete', async (req, res) => {
  const userId = req.userId!;
  const id = objectId.parse(req.params.id);

  const result = await completeTask(userId, id);
  if (!result) {
    throw new AppError(404, 'ط§ظ„ظ…ظ‡ظ…ط© ط؛ظٹط± ظ…ظˆط¬ظˆط¯ط© ط£ظˆ طھظ…طھ ط¨ط§ظ„ظپط¹ظ„', 'TASK_NOT_FOUND');
  }

  const [completed, next] = await Promise.all([
    toTaskDtos(userId, [result.completed]),
    result.next ? toTaskDtos(userId, [result.next]) : Promise.resolve([]),
  ]);
  res.json({ task: completed[0], next: next[0] ?? null });
});

import type { Types } from 'mongoose';
import { Plant } from '../models/Plant';
import { Task } from '../models/Task';
import {
  TASK_TYPES,
  addDays,
  intervalFor,
  nextDueAt,
  type CareIntervals,
  type TaskType,
} from '../utils/schedule';

interface PlantRef {
  _id: Types.ObjectId;
  userId: Types.ObjectId;
  care?: CareIntervals | null;
}

const isDuplicateKey = (e: unknown) =>
  typeof e === 'object' && e !== null && 'code' in e && e.code === 11000;

/**
 * ظٹط®ظ„ظ‘ظٹ ط§ظ„ظ…ظ‡ط§ظ… ط§ظ„ظ…ط¹ظ„ظ‘ظ‚ط© ظ…طھط·ط§ط¨ظ‚ط© ظ…ط¹ ظ…ظ„ظپ ط§ظ„ط¹ظ†ط§ظٹط©:
 * ظپطھط±ط© ط§طھط´ط§ظ„طھ -> طھطھظ…ط³ط­ ط§ظ„ظ…ظ‡ظ…ط© ط§ظ„ظ…ط¹ظ„ظ‘ظ‚ط©طŒ ظˆظپطھط±ط© ط§طھط¶ط§ظپطھ -> طھطھط¹ظ…ظ„ ظ…ظ‡ظ…ط© ط¬ط¯ظٹط¯ط©.
 * ظ…ظ‡ظ…ط© ظ…ظˆط¬ظˆط¯ط© ط¨ط§ظ„ظپط¹ظ„ ط¨طھظپط¶ظ„ ط²ظٹ ظ…ط§ ظ‡ظٹ (ط§ظ„ظپطھط±ط© ط§ظ„ط¬ط¯ظٹط¯ط© ط¨طھط·ط¨ظ‘ظ‚ ظ…ظ† ط§ظ„ظ…ظ‡ظ…ط© ط§ظ„ظ„ظٹ ط¨ط¹ط¯ظ‡ط§).
 */
export async function syncPendingTasks(
  plant: PlantRef,
  now: Date = new Date(),
): Promise<void> {
  const pending = await Task.find({ plantId: plant._id, status: 'pending' }).select(
    'type',
  );
  const have = new Set<string>(pending.map((t) => t.type));

  const toRemove: TaskType[] = [];
  const toCreate: {
    userId: Types.ObjectId;
    plantId: Types.ObjectId;
    type: TaskType;
    dueAt: Date;
  }[] = [];

  for (const type of TASK_TYPES) {
    const days = intervalFor(plant.care, type);
    if (days === null && have.has(type)) toRemove.push(type);
    if (days !== null && !have.has(type)) {
      toCreate.push({
        userId: plant.userId,
        plantId: plant._id,
        type,
        dueAt: addDays(now, days),
      });
    }
  }

  if (toRemove.length > 0) {
    await Task.deleteMany({
      plantId: plant._id,
      status: 'pending',
      type: { $in: toRemove },
    });
  }
  if (toCreate.length > 0) {
    try {
      await Task.insertMany(toCreate, { ordered: false });
    } catch (e) {
      // ط·ظ„ط¨ ظ…طھط²ط§ظ…ظ† ط³ط¨ظ‚ظ†ط§ ظˆط¹ظ…ظ„ ظ†ظپط³ ط§ظ„ظ…ظ‡ظ…ط©طŒ ظ…ط´ ظ…ط´ظƒظ„ط©
      if (!isDuplicateKey(e)) throw e;
    }
  }
}

/**
 * ظٹط¹ظ„ظ‘ظ… ط§ظ„ظ…ظ‡ظ…ط© طھظ…طھ ظˆظٹط¹ظ…ظ„ ط§ظ„ظ…ظ‡ظ…ط© ط§ظ„ظ„ظٹ ط¨ط¹ط¯ظ‡ط§.
 * ط§ظ„طھط­ط¯ظٹط« ط°ط±ظٹ (status: pending ظپظٹ ط§ظ„ظپظ„طھط±)طŒ ظپظ„ظˆ ط§طھط¨ط¹طھ ط·ظ„ط¨ظٹظ†طŒ ظˆط§ط­ط¯ ط¨ط³ ط¨ظٹظƒط³ط¨.
 */
export async function completeTask(
  userId: string,
  taskId: string,
  now: Date = new Date(),
) {
  const task = await Task.findOneAndUpdate(
    { _id: taskId, userId, status: 'pending' },
    { $set: { status: 'completed', completedAt: now } },
    { new: true },
  );
  if (!task) return null;

  const plant = await Plant.findOne({ _id: task.plantId, userId }).select('care');
  let next = null;
  if (plant) {
    const due = nextDueAt(now, plant.care, task.type);
    if (due) {
      next = await Task.create({
        userId: task.userId,
        plantId: task.plantId,
        type: task.type,
        dueAt: due,
      });
    }
  }
  return { completed: task, next };
}

interface TaskLike {
  id: string;
  type: string;
  status: string;
  dueAt: Date;
  completedAt?: Date | null;
  plantId: Types.ObjectId;
}

/** ظٹط­ظˆظ‘ظ„ ط§ظ„ظ…ظ‡ط§ظ… ظ„ط´ظƒظ„ ط§ظ„ط±ط¯ ظˆظٹط¶ظٹظپ ط§ط³ظ… ظˆطµظˆط±ط© ط§ظ„ظ†ط¨ط§طھ (ط·ظ„ط¨ ظˆط§ط­ط¯ ظ…ط´ ط·ظ„ط¨ ظ„ظƒظ„ ظ…ظ‡ظ…ط©) */
export async function toTaskDtos(userId: string, tasks: TaskLike[]) {
  const ids = [...new Set(tasks.map((t) => String(t.plantId)))];
  const plants =
    ids.length > 0
      ? await Plant.find({ _id: { $in: ids }, userId }).select('nickname imageUrl')
      : [];

export function safeTimeZone(tz: string | null | undefined): string {
  if (!tz) return 'UTC';
  try {
    new Intl.DateTimeFormat('en-US', { timeZone: tz });
    return tz;
  } catch {
    return 'UTC';
  }
}

/** ظپط±ظ‚ ط§ظ„طھظˆظ‚ظٹطھ (ط¨ط§ظ„ظ…ظٹظ„ظٹ ط«ط§ظ†ظٹط©) ط¨ظٹظ† ط§ظ„ظ…ظ†ط·ظ‚ط© ط§ظ„ط²ظ…ظ†ظٹط© ظˆUTC ط¹ظ†ط¯ ظ„ط­ط¸ط© ظ…ط¹ظٹظ†ط© */
function tzOffsetMs(timeZone: string, at: Date): number {
  const parts = new Intl.DateTimeFormat('en-US', {
    timeZone,
    hourCycle: 'h23',
    year: 'numeric',
    month: 'numeric',
    day: 'numeric',
    hour: 'numeric',
    minute: 'numeric',
    second: 'numeric',
  }).formatToParts(at);
  const get = (type: string) =>
    Number(parts.find((p) => p.type === type)?.value);
  const asUtc = Date.UTC(
    get('year'),
    get('month') - 1,
    get('day'),
    get('hour'),
    get('minute'),
    get('second'),
  );
  return asUtc - Math.floor(at.getTime() / 1000) * 1000;
}

function localMidnightUtc(timeZone: string, now: Date, dayOffset: number): Date {
  const off = tzOffsetMs(timeZone, now);
  const local = new Date(now.getTime() + off);
  const target = Date.UTC(
    local.getUTCFullYear(),
    local.getUTCMonth(),
    local.getUTCDate() + dayOffset,
  );
  // ظ†ط¹ظٹط¯ ط­ط³ط§ط¨ ط§ظ„ظپط±ظ‚ ط¹ظ†ط¯ ظ„ط­ط¸ط© ظ…ظ†طھطµظپ ط§ظ„ظ„ظٹظ„ ظ†ظپط³ظ‡ط§ (ط¨ظٹطµط­ظ‘ط­ ط£ظٹط§ظ… طھط؛ظٹظٹط± ط§ظ„طھظˆظ‚ظٹطھ ط§ظ„طµظٹظپظٹ)
  const off2 = tzOffsetMs(timeZone, new Date(target - off));
  return new Date(target - off2);
}

/** ط¨ط¯ط§ظٹط© ظˆظ†ظ‡ط§ظٹط© ط§ظ„ظٹظˆظ… ط§ظ„ط­ط§ظ„ظٹ ط¨طھظˆظ‚ظٹطھ ط§ظ„ظ…ط³طھط®ط¯ظ…طŒ ظƒظ„ط­ط¸ط§طھ UTC */
export function localDayBounds(
  timeZone: string,
  now: Date = new Date(),
): { start: Date; end: Date } {
  const tz = safeTimeZone(timeZone);
  return {
    start: localMidnightUtc(tz, now, 0),
    end: localMidnightUtc(tz, now, 1),
  };
}

async function doRefresh(): Promise<AuthResponse | null> {
  const res = await send("/auth/refresh", { method: "POST" });
  if (res.status === 401) {
    accessToken = null;
    return null;
  }

export function refreshSession(): Promise<AuthResponse | null> {
  if (!refreshing) {
    const run: () => Promise<AuthResponse | null> =
      typeof navigator !== "undefined" && navigator.locks
        ? async () => await navigator.locks.request("nabtety-refresh", doRefresh)
        : doRefresh;
    const promise = run().finally(() => {
      refreshing = null;
    });
    refreshing = promise;
    return promise;
  }
  return refreshing;
}

export async function api<T>(
  path: string,
  options: { method?: string; body?: unknown } = {},
): Promise<T> {
  const method = options.method ?? "GET";
  let res = await send(path, jsonInit(method, options.body));

  if (res.status === 401 && !path.startsWith("/auth/")) {
    const session = await refreshSession();
    if (!session) {
      onSessionLost?.();
      throw await parseError(res);
    }
    res = await send(path, jsonInit(method, options.body));
  }

  if (!res.ok) throw await parseError(res);
  if (res.status === 204) return undefined as T;
  return (await res.json()) as T;
}

}
